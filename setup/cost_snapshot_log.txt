WITH r AS ( SELECT FEATURE, CREATED_AT, TRY_TO_NUMBER(REGEXP_SUBSTR(PROVENANCE, 'tokens ([0-9]+) in', 1, 1, 'e', 1)) AS IN_TOK, TRY_TO_NUMBER(REGEXP_SUBSTR(PROVENANCE, 'tokens [0-9]+ in / ([0-9]+) out', 1, 1, 'e', 1)) AS OUT_TOK, COALESCE(TRY_TO_NUMBER(REGEXP_SUBSTR(PROVENANCE, 'out / ([0-9]+) cache_read', 1, 1, 'e', 1)), 0) AS CACHE_TOK FROM WBR_DB.ANALYTICS_ENTERPRISE.AA_AI_RUNS WHERE CREATED_AT >= DATEADD('day', -30, CURRENT_TIMESTAMP()) AND REGEXP_LIKE(PROVENANCE, '.*tokens [0-9]+ in / [0-9]+ out.*')), p AS ( SELECT FEATURE, IN_TOK, OUT_TOK, CACHE_TOK, IFF(FEATURE IN ('agent_question', 'tile_question'), ((IN_TOK - CACHE_TOK) * 1.95 + CACHE_TOK * 0.20 + OUT_TOK * 9.76) / 1e6, (IN_TOK * 1.80 + OUT_TOK * 9.00) / 1e6) AS CREDITS FROM r) SELECT FEATURE, COUNT(*) AS RUNS, ROUND(AVG(IN_TOK)) AS AVG_IN_TOKENS, ROUND(AVG(CACHE_TOK)) AS AVG_CACHE_READ, ROUND(AVG(OUT_TOK)) AS AVG_OUT_TOKENS, ROUND(AVG(CREDITS), 5) AS AI_CREDITS_PER_RUN, ROUND(AVG(CREDITS) * 2.00, 4) AS USD_PER_RUN_LIST FROM p GROUP BY 1 ORDER BY USD_PER_RUN_LIST DESC
